const SCREEN_WIDTH = 240;
const SCREEN_HEIGHT = 136;

function TIC() {
	cls(0);
	const centerX = SCREEN_WIDTH / 2;
	const centerY = SCREEN_HEIGHT / 2;
	const centerRadius = 16;
	

	for (let i = 1; i < 120; i+= 2) {
		const freq = vqts(i);
	 const col = colForFreq(freq);
		const prevI = i == 0 ? 119 : i - 1;
		const nextI = i == 119 ? 0 : i + 1;
		let prevPeakX = determinePeakX(centerX, centerRadius, prevI, vqts(prevI));
		let prevPeakY = determinePeakY(centerY, centerRadius, prevI, vqts(prevI));
		let nextPeakX = determinePeakX(centerX, centerRadius, nextI, vqts(nextI));
		let nextPeakY = determinePeakY(centerY, centerRadius, nextI, vqts(nextI));
		const peakX = determinePeakX(centerX, centerRadius, i, freq);
		const peakY = determinePeakY(centerY, centerRadius, i, freq);
		drawCurve(prevPeakX, prevPeakY, peakX, peakY, nextPeakX, nextPeakY, col);
	}
}

function determinePeakX(centerX, centerRadius, i, freq) {
	const rotated = (i / 120) * Math.PI * 2;
	return centerX + (Math.sin(rotated) * ((freq * 64) + centerRadius));
}

function determinePeakY(centerY, centerRadius, i, freq) {
	const rotated = (i / 120) * Math.PI * 2;
	return centerY + (Math.cos(rotated) * ((freq * 64) + centerRadius));
}

function drawCurve(x0, y0, x1, y1, x2, y2, col) {
	const steps = 32;
	let px = x0;
	let py = y0;
	for (let i = 0; i < steps; i++) {
		const t = i / steps;
		const u = 1 - t;
		const x = (u * u * x0) + (2 * u * t * x1) + (t * t * x2);
		const y = (u * u * y0) + (2 * u * t * y1) + (t * t * y2);
		line(px, py, x, y, col);
		px = x;
		py = y;
	}
}

function colForFreq(freq) {
	if (freq > 0.909) {
		return 1;
	}
	if (freq > 0.818) {
		return 2;
	}
	if (freq > 0.727) {
		return 3;
	}
	if (freq > 0.636) {
		return 4;
	}
	if (freq > 0.545) {
  return 5;
	}
	if (freq > 0.455) {
		return 6;
	}
	if (freq > 0.364) {
		return 7;
	}
	if (freq > 0.273) {
		return 8;
	}
	if (freq > 0.182) {
		return 9;
	}
	if (freq > 0.091) {
		return 10;
	}
	return 11;
}
