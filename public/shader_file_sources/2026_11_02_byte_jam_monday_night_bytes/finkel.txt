function lin(x,x0,x1,y0,y1){
 let t=(x-x0)/(x1-x0)
 return y0+t*(y1-y0)
}

function uf(a,b){
 return lin(rnd(),0,1,a,b)
}

function cn(a){
 return uf(-a,a)
}

function c(x){
 return ()=>x
}

class P{
 constructor(x,y,vx,vy,h,dec){
  this.x=x
  this.y=y
  this.vx=vx
  this.vy=vy
  this.h=h
  this.sh=h
  this.dec=dec
  this.fr=fr
 }
 step(){
  let dt=0.1
  this.x+=this.vx*dt
  this.y+=this.vy*dt
  if(this.dec){
  this.vx*=lin(this.h,this.sh,0,1,0.99)
  }
  this.h--
 }
}

class E{
 constructor(x,y,dir,vel,rate,h,dec=false){
  this.x=x
  this.y=y
  this.dir=dir
  this.vel=vel
  this.rate=rate
  this.h=h
  this.ps=[]
  this.nfr=1
  this.dec=dec
 }
 mirrored(){
  return new E(
   ()=>W-this.x(),
   this.y,
   ()=>-this.dir(),
   this.vel,
   this.rate,
   this.h
  )
 }
 step(){
  if(fr==this.nfr){
   this.nfr+=max(1,floor(this.rate()))
   let x=this.x()
   let y=this.y()
   let vel=this.vel()
   let dir=this.dir()+tau/2
   let vx=vel*sin(dir)
   let vy=vel*cos(dir)
   this.ps.push(new P(x,y,vx,vy,this.h(),this.dec))
  }
  this.ps.forEach(p=>{
   p.step()
  })
  this.ps=this.ps.filter(p=>p.h>0)
 }
}

function hash(f){
 return (f%5000)/5000
}
function fire(x,y,s,a){
 let x0=x-W*s*a/2
 let x1=x+W*s*a/2
 let y0=y-H*s
 let y1=y
 a=[x0,y0]
 b=[x1,y0]
 c=[x1,y1]
 d=[x0,y1]
 ua=[0,0]
 ub=[W,0]
 uc=[W,H]
 ud=[0,H]
 ttri(...a,...b,...c,...ua,...ub,...uc,2)
 ttri(...a,...c,...d,...ua,...uc,...ud,2)
}

function drive(es,sr,er,col,ecc=1){
es.forEach(e=>{
 e.ps.forEach(p=>{
  let r=lin(p.h,p.sh,0,sr,er)
  elli(p.x,p.y,r,r*ecc,col)
 })
 e.step()
 })
}
