function lin(x,x0,x1,y0,y1){
 let t =(x-x0)/(x1-x0)
 return y0+t*(y1-y0)
}

function hash(ss){
 let s=floor(ss)
 let w=((s>>((s>>20)+4))^s)*277803737
 return (w>>22)^w
}

function nhash(s){
 return (hash(s)%5000)/5000
}

function chash(s){
 return lin(nhash(s),0,1,-1,1)
}

function make_pal(){
 for(var i=0;i<8;i++){
  let cs =[
   255*(i&1?1:0),
   255*(i&2?1:0),
   255*(i&4?1:0)
  ]
  for(var j=0;j<3;j++){
   poke(PAL+i*3+j,cs[j])
  }
 }
}
