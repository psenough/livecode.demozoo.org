function track(){

class P{
 constructor(x,y){
  this.x=x
  this.y=y
  this.vx=0
  this.vy=0
  this.h=-1
 }

 step(){
  let dt=0.1
  this.x+=this.vx*dt
  this.y+=this.vy*dt
  if(this.h<0){
   this.h=uf(30,50)
   let dir = ri(0,3)
   let th=tau*dir/4
   let sc=200
   this.vx=cos(th)*sc
   this.vy=sin(th)*sc
  }
  this.h--
  this.x=(this.x+W)%W
  this.y=(this.y+H)%H
  let damp=0.8
  this.vx*=damp
  this.vy*=damp
 }
}

function make_pal(h,s){
 let mx=lin(0,1,1,2,s)
 let mi=lin(0,1,1,0.5,s)
 let r=lin(-1,1,mi,mx,sin(h))
 let g=lin(-1,1,mi,mx,sin(h+tau/3))
 let b=lin(-1,1,mi,mx,sin(h+tau/3*2))
 let en=lin(0,1,1,1.2,le)
 let flash=he>0.8&&rnd()>0.8
 range(16).flatMap(i=>{
  let x=i/15
  return [
   pow(x*en,r)*255,
   pow(x*en,g)*255,
   pow(x*en,b)*255,
  ]
 }).forEach((c,i)=>{
  let off=i+floor(fr/30+leacc/8)*3
  off=off%(16*3)
  off=off<3?i:off
  c=flash&&off>=3?uf(180,300)-c:c
  poke(PALETTE+off,c)
 })
}

function pal(){
	make_pal(time()/1000,
	lin(-1,1,0.5,1.2,sin(time()/1500)))
}

 }
