h=0
function make_pal(){
 f=fft(5)*5
 f=lin(0,1,1,0.5,f,true)
 r=lin(-1,1,.8,3,sin(h))*f
 g=lin(-1,1,.8,3,sin(h+tau/3))*f
 b=lin(-1,1,.8,3,sin(h+tau/3*2))*f
 range(16).flatMap(i=>{
  x=i/15
  return[
   255*pow(x,r),
   255*pow(x,g),
   255*pow(x,b),
  ]
 }).forEach((p,i)=>{
  poke(PALETTE+i,p)
 })
 h+=0.01
}

class DP{
 constructor(){
  this.th1=0.1
  this.th2=-0.1
  this.dth1=0
  this.dth2=0
  this.m1=uf(8,13)
  this.m2=uf(5,10)
  this.l1=uf(0.8,1.2)
  this.l2=2-this.l1
 }
 step(dt){
	 let th1=this.th1
  let th2=this.th2
  let dth1s=this.dth1*this.dth1
  let dth2s=this.dth2*this.dth2
  let m1=this.m1
  let m2=this.m2
  let l1=this.l1
  let l2=this.l2
  let g=-10

  //sorry about this :)
  let den=2*m1+m2-m2*cos(2*th1-2*th2)

  var ddth1=-g*(2*m1+m2)*sin(th1)-m2*g*sin(th1-2*th2)-2*sin(th1-th2)*m2*(dth2s*l2+dth1s*l1*cos(th1-th2))
  ddth1/=l1*den

  var ddth2=2*sin(th1-th2)*(dth1s*l1*(m1+m2)+g*(m1+m2)*cos(th1)+dth2s*l2*m2*cos(th1-th2))
  ddth2/=l2*den

  this.th1+=this.dth1*dt
  this.th2+=this.dth2*dt
  this.dth1+=ddth1*dt-this.dth1*0.01*dt
  this.dth2+=ddth2*dt-this.dth2*0.1*dt
 }
}

class P{
 constructor(x,y,vx,vy,h){
  this.x=x
  this.y=y
  this.vx=vx
  this.vy=vy
  this.h=h
  this.ax=x
  this.ay=y
 }
 step(dt){
  var distort=lin(300,0,0,1,this.h)
  //distort=pow(distort,2)
  //distort/=3
  distort*=fft(5)*20

  this.x+=this.vx*dt
  this.y+=this.vy*dt
  this.vx*=0.99//+uf(-1,1)*distort
  this.vy*=0.99//+uf(-1,1)*distort
  this.h--
  this.ax=this.x+uf(-1,1)*distort
  this.ay=this.y+uf(-1,1)*distort
 }
}
