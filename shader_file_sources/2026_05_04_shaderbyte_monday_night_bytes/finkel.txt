function track(){
 damp=clamp(0,1,time()/1000)
 le=max(le,fft(2,4)*0.5*damp)
 le*=0.97
 he=max(he,fft(200,250)*0.8*damp)
 he*=0.97
}

function nhash(t){
 return (hash(t)%1000)/1000
}

function noise1(x){
 let x0=floor(x)
 let x1=x0+1
 let xf=x-x0
 let z0=nhash(x0)
 let z1=nhash(x1)
 return lerp(z0,z1,xf)
}

function noise2(x,y){
 let y0=floor(y)
 let y1=y0+1
 let yf=y-y0
 let z0=noise1(x+y0*1234)
 let z1=noise1(x+y1*1234)
 return lerp(z0,z1,yf)
}

function fnoise(x,y,itr=7){
 var acc=0
 var amp=1
 for(var i=0;i<itr;i++){
  acc+=noise2(x,y)*amp
  amp*=lin(0,1,0.3,0.6,he,true)
  x*=2
  y*=2
 }
 return acc
}

function make_pal(){
 t=le*70
 Array.from([]).concat(
  [[200+t,1-le/3,0.3+le/4]],
  [[200+t,0.5,0.9]],
  [[180,0.1,0.9]],
  [[50,1,1-le/3]],
  [[30,1,0.4]],
  vint(6,[180+t,1,0.5],[140+t,1,0.9]),
  vint(5,[200+t,0.5+le/4,0.5+le/5],[200-t,0.1,1]),
 ).flatMap(hsv=>hsv2rgb(...hsv))
 .forEach((c,i)=>{
  poke(PALETTE+i,c)
 })
}

function vint(num,f,t){
 if (num==1){
  return [f]
 }
 return range(num).map(i=>{
  let x=i/(num-1)
  return f.map((e,j)=>lerp(e,t[j],x))
 })
}
