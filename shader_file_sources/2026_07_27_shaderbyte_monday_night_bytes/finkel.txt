function track(){
 damp=clamp(0,1,time()/1000)
 le=max(le,fft(2,4)*0.5*damp)
 le*=0.7
 he=max(he,fft(200,250)*5.5*damp)
 he*=0.9
}

function BOOT(){
 cls(0)
}

function TIC(){
	track()
	vbank(0)
	memcpy(BUFFER,0,SZ/2)
	vbank(1)
	memcpy(0,BUFFER,SZ/2)
	vbank(0)
	make_pal(0,1)
	
	con=sin(time()/1000)*10-5
	con/=3
	th=time()/500
	r=20
	ra=sin(th)*r
	rb=sin(th*2)*r
	rc=sin(th*1.5)*r
	rd=sin(th*1.3)*r
	
	a=[con+ra*sin(th*1.3),con+ra*sin(th*1.5)]
	b=[W-con+rb*sin(th),con+rb*sin(th*2)]
	c=[W-con+rc*sin(th/1.5),H-con+rc*sin(th*1.78)]
	d=[con+rd*sin(th*1.5),H-con+rd*sin(th*1.6)]
	ua=[0,0]
	ub=[W,0]
	uc=[W,H]
	ud=[0,H]
	
	let shifts=[
	 [a,b,c,d],
	 [b,c,d,a],
		[b,a,d,c],
		[d,c,b,a]
	]
	let [aa,bb,cc,dd]=shifts[floor(time()/200)%shifts.length]
	
	ttri(...aa,...bb,...cc,...ua,...ub,...uc,2)
	ttri(...aa,...cc,...dd,...ua,...uc,...ud,2)
	
	for (i=0;i<SZ;i++){
	 c=peek4(i)
		if(c>0 && rnd()>0.8){
		 poke4(i,c-1)
		}
		if(c>8 && rnd()>0.1){
		 poke4(i,c-ri(1,3))
		}
		if(c>3 && c<7 && rnd()>0.99){
		 poke4(i,15)
		}
	}
	
	circb(MW,MH,lin(0,1,40,80,le,true)+uf(-1,1)*5,15)
	circb(MW,MH,lin(0,1,5,20,le,true)+uf(-1,1)*10,15)
	vbank(1)
	cls(0)
	vbank(0)
}

function make_pal(hue,sat){
 let mi=lin(1,0,0.5,1,sat)
 let mx=lin(1,0,2,1,sat)
 let r=lin(-1,1,mi,mx,sin(hue))
 let g=lin(-1,1,mi,mx,sin(hue+tau/3))
 let b=lin(-1,1,mi,mx,sin(hue+tau/3*2))
 range(16).flatMap(i=>{
  let x=i/15
  return [
   pow(x,r)*255,
   pow(x,g)*255,
   pow(x,b)*255,
  ]
 }).forEach((c,i)=>{
  poke(PALETTE+i,c)
 })
}

function BDR(i){
 if(i%10 == 0){
  make_pal(time()/1000+i/100,1)
 }
 
}
