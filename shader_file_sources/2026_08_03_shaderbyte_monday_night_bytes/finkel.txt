class P{
 constructor(x,y){
  this.x=x
  this.y=y
  this.vx=4
  this.vy=8
 }

 step(){
  let dt=0.03

  let dx=lpx-this.x
  let dy=lpy-this.y
  this.vx+=dx*dt
  this.vy+=dy*dt
  let damp=0.99
  this.vx*=damp
  this.vy*=damp
  this.x+=this.vx*dt
  this.y+=this.vy*dt

  let bounce=-0.5
  if(this.x<0){
   this.x*=-1
   this.vx*=bounce
  }
  if(this.x>W){
   this.x=2*W-this.x
   this.vx*=bounce
  }
  if(this.y<0){
   this.y*=-1
   this.vy*=bounce
  }
  if(this.y>H){
   this.y=2*H-this.y
   this.vy*=bounce
  }
 }
 collide(o){
  let dx=o.x-this.x
  let dy=o.y-this.y
  let distsq=dx*dx+dy*dy
  if(distsq>30){
   return
  }
  let dist=sqrt(distsq)

  let sc=10

  this.vx=-dx*sc
  o.vx=dx*sc
  this.vy=-dy*sc
  o.vy=dy*sc

 }
}

function box(pts,col=12){
 tri(...pts[0],...pts[1],...pts[2],col)
 tri(...pts[0],...pts[2],...pts[3],col)
}

function rot(a){
 return v=>{
 return [cos(a)*v[0]-sin(a)*v[1],sin(a)*v[0]+cos(a)*v[1]]
 }
}

function scale(s){
 return v=>[v[0]*s,v[1]*s]
}

function add(a,b){
 return v=>[v[0]+a,v[1]+b]
}

function mul(a,b){
 return v=>[v[0]*a,v[1]*b]
}
