function track(){

fr=0
class P{
 constructor(x,y){
  this.x=x
  this.y=y
  this.vx=uf(-5,5)
  this.vy=uf(-5,5)
  this.th=0
  this.vth=0
  this.interval=ri(10,50)
  this.interval2=ri(60,100)
 }
 
 step(){
  let dt=0.1
  let x=this.x
  let y=this.y
  let t=time()/1000
  
  var ax=sin(x/30+t)+cos(y/60+t*2)
  var ay=sin(x/60+y/20+t*2)
  
  this.vx+=ax*dt
  this.vx-=this.vx*0.01
  this.x+=this.vx*dt
  this.vy+=ay*dt
  this.vy-=this.vy*0.01
  this.y+=this.vy*dt
  
  if(fr%this.interval2==0){
   this.vx+=uf(-1,1)*30
   this.vy+=uf(-1,1)*30
  }
  
  if(fr%this.interval==0){
   this.vth+=uf(-1,1)*10
  }
  this.th+=this.vth*dt
  this.vth*=0.95
  
  
  if (x<0){
   this.x*=-1
   this.vx*=-1
  }
  if (x>W){
   this.x=2*W-x
   this.vx*=-1
  }
  if (y<0){
   this.y*=-1
   this.vy*=-1
  }
  if (y>H){
   this.y=2*H-y
   this.vy*=-1
  }
 }
}

function make_pal(hue,sat){
 let t=hue
 let rmin=lin(0,1,1,0.5,sat)
 let rmax=lin(0,1,1,2,sat)
 let r=lin(-1,1,rmin,rmax,sin(t))
 let g=lin(-1,1,rmin,rmax,sin(t+tau/3))
 let b=lin(-1,1,rmin,rmax,sin(t+tau/3*2))
 range(16).flatMap(i=>{
  let x=i/15.0
  return [
   pow(x,r)*255,
   pow(x,g)*255,
   pow(x,b)*255,
  ]
 }).forEach((c,i)=>{
  poke(PALETTE+i,c)
 })
}
