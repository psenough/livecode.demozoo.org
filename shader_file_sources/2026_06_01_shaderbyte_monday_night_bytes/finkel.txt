function track(){
 let damp=clamp(0,1,time()/1000)
 let l_amp=3.0
 let l_decay=0.4
 let l_floor=0.5
 let h_amp=1.5
 let h_decay=0.85
 _le=max(_le,fft(2,4)*l_amp*damp)
 _le*=l_decay
 le=max(0,_le-l_floor)
 he=max(he,fft(200,250)*h_amp*damp)
 he*=h_decay
}

shake=0

class P{
 constructor(x,y){
  this.x=x
  this.y=y
  this.px=x
  this.py=y
  this.vx=0
  this.vy=0
  this.health=uf(20,100)
 }

 reset(){
  this.health=uf(20,100)
  let a=uf(40,120)
  let th=uf(0,tau)
  this.vx=a*sin(th)
  this.vy=a*cos(th)
  shake+=1
 }

 step(){
  let dt=0.1
  this.px=this.x
  this.py=this.y
  this.x+=this.vx*dt
  this.y+=this.vy*dt

  let t=time()/1000
  let x=this.x
  let y=this.y
  let sc=lin(0,100,10,40,this.health)
  this.vx+=sin((x+y*sin(t)*2)/sc)
  this.vy+=sin((x/2*cos(t*2)+y*sin(t))/sc)

  this.vx-=(x-MW+sin(t/4+this.health/30)*MW/3)/30

  let damp=0.9

  this.vx*=damp
  this.vy*=damp

  if(x<0){
   this.x*=-1
   this.vx*=-1
  }
  if(x>W){
   this.x=2*W-this.x
   this.vx*=-1
  }
  if(y<0){
   this.y*=-1
   this.vy*=-1
  }
  if(y>H){
   this.y=2*H-this.y
   this.vy*=-1
  }

  this.health-=le
  if(this.health<0){
   this.reset()
  }
 }
}

   pow(x,r)*255,
