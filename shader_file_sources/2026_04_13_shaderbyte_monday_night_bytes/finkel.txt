class P{
 constructor(x,y){
  this.x=x
  this.y=y
  this.ox=0
  this.oy=0
 }
 pos(t){
  return [
   this.x+this.ox*t,
   this.y+this.oy*t,
  ]
 }
 reset(){
  this.x+=this.ox
  this.y+=this.oy
  this.x=(this.x+nx)%nx
  this.y=(this.y+ny)%ny
  this.ox=0
  this.oy=0
 }
}

function reset(){
 vert=pick(true,false)
 n=vert?nx:ny
 dirs=range(n).map(_=>pick(-2,-1,-1,0,1,1,2))
 ps.forEach(p=>{
  p.reset()
  if (vert){
   p.oy=dirs[p.x]
  }
  else{
   p.ox=dirs[p.y]
  }
 })
}

})

function make_pal(){
