class P{
 constructor(x,y,vx,vy,health,gen){
  this.x=x
  this.y=y
  this.vx=vx
  this.vy=vy
  this.health=health
  this.gen=gen
 }
 step(dt){
  this.x+=this.vx*dt
  this.y+=this.vy*dt
  this.vy+=-10*dt
  if (this.x<0){
   this.x*=-1
   this.vx*=-1
  }
  if(this.x >W){
   this.x=2*W-this.x
   this.vx*=-1
  }
  if (this.y<0){
   this.y*=-1
   this.vy*=-1
  }
  if(this.y >H){
   this.y=2*H-this.y
   this.vy*=-1
  }
  this.health-=fft(5)*10
 }
}

function make_palette(){
 r=uf(0.1,2)
 g=uf(0.1,2)
 b=uf(0.1,2)
 range(16).flatMap((i) => {
  x=i/16.0
  return [
   255*pow(x,r),
   255*pow(x,g),
   255*pow(x,b),
  ]
 }).forEach((v,i)=>{
  poke(PALETTE+i,v)
 })
}
