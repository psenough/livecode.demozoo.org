    function sinc(x)
{
	return (x == 0.0) ? 1.0 : Math.sin(x*Math.PI)/(x*Math.PI);   
}

//https://en.wikipedia.org/wiki/Window_function
function WindowBlackman(a, N, i)
{
    var a0 = (1.0 - a) / 2.0;
    var a1 = 0.5;
    var a2 = a / 2.0;
    
    var wnd = a0;
    wnd -= a1 * Math.cos(2.0 * Math.PI * ((i) / (N - 1.0)));
    wnd += a2 * Math.cos(4.0 * Math.PI * ((i) / (N - 1.0)));
    
    return wnd;
}
//FIR lowpass filter 
//Fc = Cutoff freq., Fs = Sample freq., N = # of taps, i = Tap index
function Lowpass( Fc,  Fs,  N,  i)
{    
    var wc = (Fc/Fs);
    
    var wnd = WindowBlackman(0.16, N, i);
    
    return 2*wc * wnd * sinc(2.0*wc * (i - N/2.0));
}
