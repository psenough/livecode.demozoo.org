function track(){

 le=max(le,fft(2,4)*0.8*damp)

 he*=0.85
}

function cstack(ox,oy,s){
 c=20
 for(i=0;i<c;i++){
  r=s*(c-i)+((i%2)-0.5)*lin(0,1,-5,8,le)+uf(-1,1)*(he*4+0.5)
  var x=MW+ox
  var y=MH+oy
  circ(x,y,r,(i/2)*(i%2))
 }
}

function make_pal(){
	t=time()/3
	a=he>1.1
	a=a?pick(true,false):false
 range(16).flatMap(i=>{
  if(i==0){
	  return hsv2rgb(t,1,0.2)
  }
  off=0
  if(le>0.6){
   off=pick(0,0,0,180)
  }
  x=i/15
  return hsv2rgb(
   t+rnd()*200+off,
   lerp(0.3,1,x),
   1,
  )
 }).forEach((c,i)=>{
  poke(PALETTE+i,a?255-c:c)
 })
}

function BDR(){
